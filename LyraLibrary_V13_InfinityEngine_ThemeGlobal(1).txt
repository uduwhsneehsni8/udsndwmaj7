local SetProps, SetChildren, InsertTheme, Create



function InsertTheme(Instance, Type)

    table.insert(redzlib.Instances, {Instance = Instance, Type = Type})

    return Instance

end



function SetChildren(Instance, Children)

    if Children then

        for _, Child in pairs(Children) do

            Child.Parent = Instance

        end

    end

    return Instance

end



function SetProps(Instance, Props)

    if Props then

        for prop, value in pairs(Props) do

            Instance[prop] = value

        end

    end

    return Instance

end



Create = function(...)

    local args = {...}

    local new = Instance.new(args[1])

    if type(args[2]) == "table" then

        SetProps(new, args[2])

        SetChildren(new, args[3])

    elseif typeof(args[2]) == "Instance" then

        new.Parent = args[2]

        SetProps(new, args[3])

        SetChildren(new, args[4])

    end

    return new

end

end



local function CreateTween(Configs)

    local Instance = Configs[1] or Configs.Instance

    local Prop = Configs[2] or Configs.Prop

    local NewVal = Configs[3] or Configs.NewVal

    local Time = Configs[4] or Configs.Time or 0.5

    local TweenWait = Configs[5] or Configs.wait or false

    local Tween = TweenService:Create(Instance, TweenInfo.new(Time, Enum.EasingStyle.Quint), {[Prop] = NewVal})

    Tween:Play()

    if TweenWait then Tween.Completed:Wait() end

    return Tween

end



-- Lyra UI FX: helpers reutilizaveis e leves para componentes.
local function AddSoftStroke(target, color, thickness, transparency, name)
    local stroke = Instance.new("UIStroke")
    stroke.Name = name or "LyraSoftStroke"
    stroke.Color = color or Color3.fromRGB(155, 65, 245)
    stroke.Thickness = thickness or 1
    stroke.Transparency = transparency or 0.7
    stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    stroke.Parent = target
    return stroke
end

local function AddPressScale(target, pressedScale)
    local scale = target:FindFirstChild("LyraPressScale")
    if not scale then
        scale = Instance.new("UIScale")
        scale.Name = "LyraPressScale"
        scale.Scale = 1
        scale.Parent = target
    end
    local down = pressedScale or 0.975
    target.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            TweenService:Create(scale, TweenInfo.new(0.11, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Scale = down}):Play()
        end
    end)
    target.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            TweenService:Create(scale, TweenInfo.new(0.22, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Scale = 1}):Play()
        end
    end)
    return scale
end

local function MakeDrag(Instance)

    task.spawn(function()

        SetProps(Instance, {Active = true, AutoButtonColor = false})

        local DragStart, StartPos, InputOn

        local function Update(Input)

            local delta = Input.Position - DragStart

            local Position = UDim2.new(StartPos.X.Scale, StartPos.X.Offset + delta.X / UIScale, StartPos.Y.Scale, StartPos.Y.Offset + delta.Y / UIScale)

            CreateTween({Instance, "Position", Position, 0.35})

        end

        Instance.MouseButton1Down:Connect(function() InputOn = true end)

        Instance.InputBegan:Connect(function(Input)

            if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then

                StartPos = Instance.Position

                DragStart = Input.Position

                while UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) do

                    RunService.Heartbeat:Wait()

                    if InputOn then Update(Input) end

                end

                InputOn = false

            end

        end)

    end)

    return Instance

end



local function ConnectSave(Instance, func)

    Instance.InputBegan:Connect(function(Input)

        if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then

            while UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) do task.wait() end

        end

        func()

    end)

end

local function VerifyTheme(Theme)

    for name in pairs(redzlib.Themes) do

        if name == Theme then return true end

    end

    return false

end



local function GetColor(Instance)
    if not Instance then return "" end
    if Instance:IsA("Frame") or Instance:IsA("TextButton") or Instance:IsA("ImageButton") then
        return "BackgroundColor3"
    elseif Instance:IsA("ImageLabel") then
        return "ImageColor3"
    elseif Instance:IsA("TextLabel") or Instance:IsA("TextBox") then
        return "TextColor3"
    elseif Instance:IsA("UIStroke") then
        return "Color"
    end
    return ""
end

local Connections, Connection = {}, redzlib.Connection



local function NewConnectionList(List)

    for _, CoName in ipairs(List) do

        local ConnectedFuncs, Connect = {}, {}

        Connection[CoName] = Connect

        Connections[CoName] = ConnectedFuncs

        Connect.Name = CoName

        function Connect:Connect(func)

            if type(func) == "function" then

                table.insert(ConnectedFuncs, func)

                local Connected = {Function = func, Connected = true}

                function Connected:Disconnect()

                    if self.Connected then

                        table.remove(ConnectedFuncs, table.find(ConnectedFuncs, self.Function))

                        self.Connected = false

                    end

                end

                return Connected

            end

        end

    end

end



function Connection:FireConnection(CoName, ...)

    local ConnectionTable = type(CoName) == "string" and Connections[CoName] or Connections[CoName.Name]

    for _, Func in pairs(ConnectionTable or {}) do

        task.spawn(Func, ...)

    end

end

local Funcs = {}

function Funcs:InsertCallback(tab, func)

    if type(func) == "function" then table.insert(tab, func) end

    return func

end

function Funcs:FireCallback(tab, ...)

    for _, v in ipairs(tab) do

        if type(v) == "function" then task.spawn(v, ...) end

    end

end

function Funcs:ToggleVisible(Obj, Bool) Obj.Visible = Bool ~= nil and Bool or Obj.Visible end

function Funcs:ToggleParent(Obj, Parent, Bool)

    if Bool ~= nil then Obj.Parent = Bool

    else Obj.Parent = not Obj.Parent and Parent end

end

function Funcs:GetCallback(Configs, index)

    local func = Configs[index] or Configs.Callback or function() end

    if type(func) == "table" then return ({function(Value) func[1][func[2]] = Value end}) end

    return {func}

end

AddEle("Button", function(parent, props, ...)
    local args = {...}

    local New = InsertTheme(SetProps(Create("TextButton", parent, {
        Text = "",
        Size = UDim2.fromScale(1, 1),
        BackgroundColor3 = Theme["Color Element"],
        AutoButtonColor = false,
        BackgroundTransparency = 0.25
    }), props), "Frame")

    Make("Stroke", New, {Color = Theme["Color Element Stroke"]})
    Make("Corner", New)

    New.MouseEnter:Connect(function()
        New.BackgroundTransparency = 0.1
    end)

    New.MouseLeave:Connect(function()
        New.BackgroundTransparency = 0.25
    end)

    local ButtonScale = Instance.new("UIScale")
    ButtonScale.Name = "LyraReusableButtonScale"
    ButtonScale.Scale = 1
    ButtonScale.Parent = New

    local ButtonPressTween
    local ButtonReleaseTween
    local ButtonHover = false
    local function PlayButtonScale(target, duration, style, direction)
        if ButtonPressTween then ButtonPressTween:Cancel() end
        if ButtonReleaseTween then ButtonReleaseTween:Cancel() end
        local tween = TweenService:Create(
            ButtonScale,
            TweenInfo.new(duration, style or Enum.EasingStyle.Quad, direction or Enum.EasingDirection.Out),
            {Scale = target}
        )
        if target < 1 then ButtonPressTween = tween else ButtonReleaseTween = tween end
        tween:Play()
        return tween
    end

    New.MouseEnter:Connect(function()
        ButtonHover = true
        TweenService:Create(New, TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            BackgroundTransparency = 0.10
        }):Play()
        PlayButtonScale(1.025, 0.18, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
    end)
    New.MouseLeave:Connect(function()
        ButtonHover = false
        TweenService:Create(New, TweenInfo.new(0.20, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            BackgroundTransparency = 0.25
        }):Play()
        PlayButtonScale(1, 0.24, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
    end)

    local function PressButton()
        PlayButtonScale(0.965, 0.09, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
    end
    local function ReleaseButton()
        local release = PlayButtonScale(1.035, 0.20, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
        release.Completed:Connect(function()
            if New and New.Parent then
                PlayButtonScale(1, 0.13, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
            end
        end)
    end

    New.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then

function redzlib:GetIcon(index)

    if type(index) ~= "string" or index:find("rbxassetid://") or #index == 0 then return index end

    index = string.lower(index):gsub("lucide", ""):gsub("-", "")

    if self.Icons[index] then return self.Icons[index] end

    for Name, Icon in pairs(self.Icons) do

        if Name == index then return Icon end

    end

    return index

end



function redzlib:SetTheme(NewTheme)
    if not VerifyTheme(NewTheme) then return false end

    redzlib.Save.Theme = NewTheme
    SaveJson("Lyra library V1.json", redzlib.Save)
    Theme = redzlib.Themes[NewTheme]
    Connection:FireConnection("ThemeChanging", NewTheme)

    -- Atualiza objetos registrados e limpa referências de componentes destruídos.
    local alive = {}
    for _, entry in ipairs(redzlib.Instances) do
        local obj = entry and entry.Instance
        if obj and obj.Parent then
            table.insert(alive, entry)
            local role = entry.Type
            pcall(function()
                if role == "Gradient" then
                    obj.Color = Theme["Color Hub 1"]
                elseif role == "Frame" then
                    obj.BackgroundColor3 = Theme["Color Hub 2"]
                elseif role == "Stroke" then
                    local prop = GetColor(obj)
                    if prop ~= "" then obj[prop] = Theme["Color Stroke"] end
                elseif role == "Theme" or role == "ScrollBar" then
                    local prop = GetColor(obj)
                    if prop ~= "" then obj[prop] = Theme["Color Theme"] end
                elseif role == "Text" then
                    local prop = GetColor(obj)
                    if prop ~= "" then obj[prop] = Theme["Color Text"] end
                    if obj:IsA("TextBox") then obj.PlaceholderColor3 = Theme["Color Dark Text"] end
                elseif role == "DarkText" then
                    local prop = GetColor(obj)
                    if prop ~= "" then obj[prop] = Theme["Color Dark Text"] end
                end
            end)
        end
    end
    redzlib.Instances = alive
    Connection:FireConnection("ThemeChanged", NewTheme)
    return true
end

function redzlib:AddTheme(Name, Palette)
    if type(Name) ~= "string" or Name == "" or type(Palette) ~= "table" then return false end
    local required = {"Color Hub 1", "Color Hub 2", "Color Stroke", "Color Theme", "Color Text", "Color Dark Text", "Color Element", "Color Element Stroke"}
    for _, key in ipairs(required) do
        if Palette[key] == nil then return false end
    end
    redzlib.Themes[Name] = Palette
    return true
end

        local function BuildTabPerimeter(w, h)
            local inset = 1.0
            local left, right = inset, w - inset
            local top, bottom = inset, h - inset
            -- Make the center line follow the pill-shaped UICorner used by the tab.
            local radius = math.min(12.0, (h - inset * 2) / 2, (w - inset * 2) / 2)
            local horizontal = math.max(0, right - left - 2 * radius)
            local vertical = math.max(0, bottom - top - 2 * radius)
            local segments = {}
            local total = 0
            local function line(x1, y1, x2, y2)
                local length = math.sqrt((x2-x1)^2 + (y2-y1)^2)
                if length > 0.001 then
                    segments[#segments+1] = {kind="line", x1=x1, y1=y1, x2=x2, y2=y2, len=length}
                    total = total + length
                end
            end
            local function arc(cx, cy, a1, a2)
                local length = radius * math.abs(a2-a1)
                if length > 0.001 then
                    segments[#segments+1] = {kind="arc", cx=cx, cy=cy, r=radius, a1=a1, a2=a2, len=length}
                    total = total + length
                end
            end
            line(left+radius, top, right-radius, top)
            arc(right-radius, top+radius, -math.pi/2, 0)
            line(right, top+radius, right, bottom-radius)
            arc(right-radius, bottom-radius, 0, math.pi/2)
            line(right-radius, bottom, left+radius, bottom)
            arc(left+radius, bottom-radius, math.pi/2, math.pi)
            line(left, bottom-radius, left, top+radius)
            arc(left+radius, top+radius, math.pi, math.pi*1.5)
            return segments, total
        end

        local function PointOnPerimeter(segments, total, distance)
            distance = distance % total
            local consumed = 0
            for _, segment in ipairs(segments) do
                if distance <= consumed + segment.len then
                    local d = math.clamp((distance - consumed) / segment.len, 0, 1)
                    if segment.kind == "line" then
                        local dx, dy = segment.x2-segment.x1, segment.y2-segment.y1
                        return segment.x1 + dx*d, segment.y1 + dy*d
                    else
                        local angle = segment.a1 + (segment.a2-segment.a1)*d
                        return segment.cx + math.cos(angle) * segment.r, segment.cy + math.sin(angle) * segment.r
                    end
                end
                consumed = consumed + segment.len
            end
            return 0, 0
        end

        -- Um único ciclo por frame: mantém o reflexo visível e sincronizado em todas as abas.
        -- O segmento é curto, fino e segue a tangente da curva nos cantos arredondados.
        task.spawn(function()
            while TabSelect and TabSelect.Parent and TabShine and TabShine.Parent do
                local scale = math.max(0.01, UIScale)
                local absSize = TabSelect.AbsoluteSize
                local w, h = absSize.X / scale, absSize.Y / scale
                if w > 20 and h > 12 then
                    local segments, total = BuildTabPerimeter(w, h)
                    if total > 0 then
                        local speed = total / LyraShineInterval
                        local distance = ((os.clock() - TabShineSyncStart) * speed) % total
                        local x, y = PointOnPerimeter(segments, total, distance)
                        local bx, by = PointOnPerimeter(segments, total, distance - 2.5)
                        local ax, ay = PointOnPerimeter(segments, total, distance + 2.5)
                        local angle = math.deg(math.atan2(ay - by, ax - bx))
                        TabShine.Position = UDim2.fromOffset(x, y)
                        TabShine.Rotation = angle
                        TabShine.Size = UDim2.fromOffset(5.5, 1.1)
                        TabShine.BackgroundTransparency = 0.08
                    end
                end
                task.wait(0.025)
            end
        end)

                Selected = MultiSelect and {} or (CheckFlag(Flag) and GetFlag(Flag) or Default[1])

                

                if MultiSelect and CheckFlag(Flag) then

                    local saved = GetFlag(Flag)

                    for k, v in pairs(saved) do if type(k) == "string" then Selected[k] = v end end

                elseif MultiSelect then

                    for _, v in pairs(Default) do Selected[v] = true end

                end

                

                local function CallbackSelected()

                    SetFlag(Flag, MultiSelect and Selected or tostring(Selected))

                    Funcs:FireCallback(Callback, Selected)

                end

                

                local function UpdateLabel()

                    if MultiSelect then

                        local list = {}

                        for k, v in pairs(Selected) do if v then table.insert(list, k) end end

                        ActiveLabel.Text = (#list > 0 and table.concat(list, ", ") or "...")

                    else

                        ActiveLabel.Text = tostring(Selected or "...")

                    end

                end

                

                local function UpdateSelected()

                    for _, v in pairs(Options) do

                        local Slt = MultiSelect and v.Stats or (v.Value == Selected)

                        CreateTween({v.nodes[2], "BackgroundTransparency", Slt and 0 or (MultiSelect and 0.8 or 1), 0.35})

                        CreateTween({v.nodes[2], "Size", Slt and UDim2.fromOffset(4, MultiSelect and 12 or 14) or UDim2.fromOffset(4, 4), 0.35})

                        CreateTween({v.nodes[3], "TextTransparency", Slt and 0 or 0.4, 0.35})

                    end

                    UpdateLabel()

                end

                

                local function Select(Option)

                    if MultiSelect then

                        Option.Stats = not Option.Stats

                        Selected[Option.Name] = Option.Stats

                    else

                        Selected = Option.Value

                    end

                    CallbackSelected()

                    UpdateSelected()

                end

                

                AddOption = function(index, Value)

                    local Name = tostring(type(index) == "string" and index or Value)

                    if Options[Name] then return end

                    Options[Name] = {index = index, Value = Value, Name = Name, Stats = false}

                    if MultiSelect then Options[Name].Stats = Selected[Name] or false end

                    

                    local Btn = Make("Button", ScrollFrame, {Name = "Option", Size = UDim2.new(1, 0, 0, 21), Position = UDim2.new(0, 0, 0.5), AnchorPoint = Vector2.new(0, 0.5)})

                    local IsSelected = InsertTheme(Create("Frame", Btn, {Position = UDim2.new(0, 1, 0.5), Size = UDim2.new(0, 4, 0, 4), BackgroundColor3 = Theme["Color Theme"], BackgroundTransparency = 1, AnchorPoint = Vector2.new(0, 0.5)}), "Theme")

                    Make("Corner", IsSelected, UDim.new(0.5, 0))

                    local OptionName = InsertTheme(Create("TextLabel", Btn, {Size = UDim2.new(1, 0, 1), Position = UDim2.new(0, 10), Text = Name, TextColor3 = Theme["Color Text"], Font = Enum.Font.Gotham, TextXAlignment = "Left", BackgroundTransparency = 1, TextTransparency = 0.4, RichText = true}), "Text")

                    Btn.Activated:Connect(function() Select(Options[Name]) end)

                    Options[Name].nodes = {Btn, IsSelected, OptionName}

                end

                

                RemoveOption = function(index, Value)

                    local Name = tostring(type(index) == "string" and index or Value)

                    if Options[Name] then

                        if MultiSelect then Selected[Name] = nil else Selected = nil end

                        Options[Name].nodes[1]:Destroy()

                        Options[Name] = nil

                    end

                end

                

                GetOptions = function() return Options end

                AddNewOptions = function(List, Clear)

                    if Clear then for _, Opt in pairs(Options) do RemoveOption(Opt.index, Opt.Value) end end

                    for idx, val in pairs(List) do AddOption(idx, val) end

                    CallbackSelected()

                    UpdateSelected()

                end

                

                for idx, val in pairs(DOptions) do AddOption(idx, val) end

                CallbackSelected()

                UpdateSelected()

            end

            

            Button.Activated:Connect(Minimize)

            NoClickFrame.MouseButton1Down:Connect(Disable)

            NoClickFrame.MouseButton1Click:Connect(Disable)

            MainFrame:GetPropertyChangedSignal("Visible"):Connect(Disable)

            SelectedFrame:GetPropertyChangedSignal("AbsolutePosition"):Connect(CalculatePos)

            Button.Activated:Connect(CalculateSize)

            ScrollFrame.ChildAdded:Connect(CalculateSize)

            ScrollFrame.ChildRemoved:Connect(CalculateSize)

            CalculatePos()

            CalculateSize()

            

            local Dropdown = {}

            function Dropdown:Visible(...) Funcs:ToggleVisible(Button, ...) end

            function Dropdown:Destroy() Button:Destroy() end

            function Dropdown:Callback(...) Funcs:InsertCallback(Callback, ...)(Selected) end

            function Dropdown:Add(...)

                local NewOptions = {...}

                if type(NewOptions[1]) == "table" then for _, Name in pairs(NewOptions[1]) do AddOption(Name, Name) end

                else for _, Name in pairs(NewOptions) do AddOption(Name, Name) end end

            end

            function Dropdown:Remove(Option)

                for idx, Val in pairs(GetOptions()) do

                    if (type(Option) == "number" and idx == Option) or Val.Name == Option then RemoveOption(idx, Val.Value) end

                end

            end

            function Dropdown:Select(Option)

                for _, Val in pairs(GetOptions()) do if Val.Name == Option or Val.Value == Option then Select(Val) end end

            end

            function Dropdown:Set(Val1, Clear)

                if type(Val1) == "table" then AddNewOptions(Val1, not Clear)

                elseif type(Val1) == "function" then Callback = Val1 end

            end

            return Dropdown

        end

                local Number = tonumber(NewValue * Increase)

                Number = math.floor(Number * 100) / 100

                Default, LabelVal.Text = Number, tostring(Number)

            end

            

            local function ControlPos()

                local MousePos = Player:GetMouse()

                local APos = MousePos.X - BaseMousePos.AbsolutePosition.X

                local ConfigureDpiPos = APos / SliderBar.AbsoluteSize.X

                SliderIcon.Position = UDim2.new(math.clamp(ConfigureDpiPos, 0, 1), 0, 0.5, 0)

            end

            

            local function UpdateValues()

                Indicator.Size = UDim2.new(SliderIcon.Position.X.Scale, 0, 1, 0)

                local SliderPos = SliderIcon.Position.X.Scale

                local NewValue = math.floor(((SliderPos * Max) / Max) * (Max - Min) + Min)

                UpdateLabel(NewValue)

            end

            

            SliderHolder.MouseButton1Down:Connect(function()

                CreateTween({SliderIcon, "Transparency", 0, 0.3})

                Container.ScrollingEnabled = false

                while UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) do

                    task.wait()

                    ControlPos()

                end

                CreateTween({SliderIcon, "Transparency", 0.2, 0.3})

                Container.ScrollingEnabled = true

            end)

                CreateTween({UIScaleObj, "Scale", 1, 0.2})

            end)

            

            local function SetSlider(NewValue)

                if type(NewValue) ~= "number" then return end

                local MinVal, MaxVal = Min * Increase, Max * Increase

                local SliderPos = (NewValue - MinVal) / (MaxVal - MinVal)

                SetFlag(Flag, NewValue)

                CreateTween({SliderIcon, "Position", UDim2.fromScale(math.clamp(SliderPos, 0, 1), 0.5), 0.3, true})

            end

            SetSlider(Default)

            SliderIcon:GetPropertyChangedSignal("Position"):Connect(UpdateValues)

            UpdateValues()

                if NewVal1 and NewVal2 then LabelFunc:SetTitle(NewVal1); LabelFunc:SetDesc(NewVal2)

                elseif type(NewVal1) == "string" then LabelFunc:SetTitle(NewVal1)

                elseif type(NewVal1) == "function" then Callback = NewVal1

                elseif type(NewVal1) == "number" then SetSlider(NewVal1) end

            function Slider:Visible(...) Funcs:ToggleVisible(Button, ...) end

            function Slider:Destroy() Button:Destroy() end

            return Slider

        end

        function Tab:AddColorPicker(Configs)
            Configs = Configs or {}
            local title = Configs.Name or Configs.Title or Configs[1] or "Color"
            local current = typeof(Configs.Default) == "Color3" and Configs.Default or Color3.fromRGB(160, 55, 245)

            })
            local preview = Create("Frame", holder, {
                Size = UDim2.fromOffset(28, 22), Position = UDim2.new(1, -112, 0.5, 0),
                AnchorPoint = Vector2.new(0, 0.5), BackgroundColor3 = current, BorderSizePixel = 0
            })
            Make("Corner", preview, UDim.new(0, 6))
            local previewStroke = Instance.new("UIStroke")

            local input = Create("TextBox", holder, {
                Size = UDim2.fromOffset(72, 24), Position = UDim2.new(1, -8, 0.5, 0),
                AnchorPoint = Vector2.new(1, 0.5), BackgroundColor3 = Theme["Color Element"],
                BackgroundTransparency = 0.1, Text = string.format("#%02X%02X%02X",
                    math.floor(current.R*255), math.floor(current.G*255), math.floor(current.B*255)),
                TextColor3 = Theme["Color Text"], PlaceholderText = "#RRGGBB",
                Font = Enum.Font.GothamBold, TextSize = 9, ClearTextOnFocus = false, BorderSizePixel = 0
            })
            Make("Corner", input, UDim.new(0, 6))
            local function parseColor()
                local hex = input.Text:gsub("#","")
                if #hex == 6 and hex:match("^%x%x%x%x%x%x$") then
                    local nr = tonumber(hex:sub(1,2),16)
                    local ng = tonumber(hex:sub(3,4),16)
                    local nb = tonumber(hex:sub(5,6),16)
                    current = Color3.fromRGB(nr,ng,nb)
                    preview.BackgroundColor3 = current
                    task.spawn(callback, current)
                    input.Text = string.format("#%02X%02X%02X",nr,ng,nb)
                else
                    input.Text = string.format("#%02X%02X%02X",
                        math.floor(current.R*255),math.floor(current.G*255),math.floor(current.B*255))
                end
            end
            input.FocusLost:Connect(parseColor)
            local object = {}
            function object:Set(color)
                if typeof(color) ~= "Color3" then return end
                current = color; preview.BackgroundColor3 = color
                input.Text = string.format("#%02X%02X%02X",math.floor(color.R*255),math.floor(color.G*255),math.floor(color.B*255))
                task.spawn(callback,color)
            end
            function object:Get() return current end
            function object:Destroy() holder:Destroy() end
            function object:Visible(v) holder.Visible = v ~= false end
            return object
        end

        local function LyraHoverScale(control, scaleObject, stroke)
            if not control or not control:IsA("GuiObject") then return end
            control.MouseEnter:Connect(function()
                if scaleObject and scaleObject.Parent then
                    TweenService:Create(scaleObject,TweenInfo.new(0.18,Enum.EasingStyle.Back,Enum.EasingDirection.Out),{Scale=1.035}):Play()
                end
                if stroke then TweenService:Create(stroke,TweenInfo.new(0.16),{Transparency=0.32}):Play() end
            end)
            control.MouseLeave:Connect(function()
                if scaleObject and scaleObject.Parent then
                    TweenService:Create(scaleObject,TweenInfo.new(0.2,Enum.EasingStyle.Quad,Enum.EasingDirection.Out),{Scale=1}):Play()
                end
                if stroke then TweenService:Create(stroke,TweenInfo.new(0.18),{Transparency=0.72}):Play() end
            end)
            control.InputBegan:Connect(function(input)
                if input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then
                    if scaleObject and scaleObject.Parent then TweenService:Create(scaleObject,TweenInfo.new(0.08,Enum.EasingStyle.Quad,Enum.EasingDirection.Out),{Scale=0.965}):Play() end
                end
            end)
            control.InputEnded:Connect(function(input)
                if input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then
                    if scaleObject and scaleObject.Parent then TweenService:Create(scaleObject,TweenInfo.new(0.2,Enum.EasingStyle.Back,Enum.EasingDirection.Out),{Scale=1.025}):Play() end
                end
            end)
        end

    BuildThemeStudio = function()
        local studio = Window:MakeTab({"Theme Studio"})
        studio:AddSection({"APARÊNCIA"})
        studio:AddParagraph({"Lyra Theme Studio", "Personalize cores, transparência, brilho e animações em tempo real."})
        studio:AddDropdown({
            Name = "Tema predefinido",
            Options = (function() local names = {}; for name in pairs(redzlib.Themes) do table.insert(names, name) end; table.sort(names); return names end)(),
            Default = redzlib.Save.Theme or "DarkPurple",
            Callback = function(value)
                if redzlib.Themes[value] then redzlib:SetTheme(value) end
            end
        })
        studio:AddSection({"COR PERSONALIZADA"})
        studio:AddColorPicker({
            Name = "Cor principal (HEX)",
            Default = Color3.fromRGB(160, 55, 245),
            Callback = function(color) redzlib:SetAccentColor(color) end
        })
        local r, g, b = 160, 55, 245
        local function applyAccent() redzlib:SetAccentColor(Color3.fromRGB(r,g,b)) end
        studio:AddSlider({Name = "Vermelho (R)", Min = 0, Max = 255, Increase = 1, Default = r, Callback = function(v) r = math.floor(v); applyAccent() end})
        studio:AddSlider({Name = "Verde (G)", Min = 0, Max = 255, Increase = 1, Default = g, Callback = function(v) g = math.floor(v); applyAccent() end})
        studio:AddSlider({Name = "Azul (B)", Min = 0, Max = 255, Increase = 1, Default = b, Callback = function(v) b = math.floor(v); applyAccent() end})
        studio:AddSection({"EFEITOS VISUAIS"})
        studio:AddSlider({Name = "Transparência do fundo", Min = 0, Max = 0.35, Increase = 0.01, Default = 0.07, Callback = function(v)
            if BackgroundImage and BackgroundImage.Parent then BackgroundImage.ImageTransparency = math.clamp(v,0,0.35) end
        end})
        studio:AddSlider({Name = "Intensidade do neon", Min = 0, Max = 100, Increase = 1, Default = 44, Callback = function(v)
            if MainNeonGlow and MainNeonGlow.Parent then MainNeonGlow.Transparency = 1-math.clamp(v/100,0,1) end
        end})
        studio:AddSlider({Name = "Intervalo do brilho (segundos)", Min = 1, Max = 8, Increase = 1, Default = 3, Callback = function(v)
            LyraShineInterval = math.clamp(v,1,8)
        end})
        studio:AddToggle({Name = "Partículas cósmicas", Default = true, Callback = function(enabled)
            LyraParticlesEnabled = enabled
            for _, particle in ipairs(LyraParticles) do if particle and particle.Parent then particle.Visible = enabled end end
        end})
        studio:AddToggle({Name = "Borda neon", Default = true, Callback = function(enabled)
            if MainNeonGlow and MainNeonGlow.Parent then MainNeonGlow.Enabled = enabled end
        end})
        studio:AddSection({"INFINITY THEME STUDIO"})
        studio:AddToggle({Name = "Animações de interface", Default = true, Callback = function(enabled)
            LyraParticlesEnabled = enabled
            for _, particle in ipairs(LyraParticles) do if particle and particle.Parent then particle.Visible = enabled end end
        end})
        studio:AddToggle({Name = "Reduzir movimento", Default = false, Callback = function(enabled)
            if enabled then
                LyraParticlesEnabled = false
                for _, particle in ipairs(LyraParticles) do if particle and particle.Parent then particle.Visible = false end end
                LyraShineInterval = 8
            else
                LyraParticlesEnabled = true
                LyraShineInterval = 3
            end
        end})
        studio:AddSection({"NEBULA VISUAL"})
        studio:AddToggle({Name = "Efeito de partículas", Default = true, Callback = function(enabled)
            LyraParticlesEnabled = enabled
            for _, particle in ipairs(LyraParticles) do if particle and particle.Parent then particle.Visible = enabled end end
        end})
        studio:AddToggle({Name = "Modo Glass", Default = true, Callback = function(enabled)
            if MainFrame and MainFrame.Parent then MainFrame.BackgroundTransparency = enabled and 0.12 or 0 end
            if BackgroundImage and BackgroundImage.Parent then BackgroundImage.ImageTransparency = enabled and 0.07 or 0.28 end
        end})
        studio:AddToggle({Name = "Modo reduzir efeitos", Default = false, Callback = function(enabled)
            LyraParticlesEnabled = not enabled
            for _, particle in ipairs(LyraParticles) do if particle and particle.Parent then particle.Visible = not enabled end end
            LyraShineInterval = enabled and 8 or 3
        end})
        studio:AddSection({"FERRAMENTAS DO TEMA"})
        studio:AddButton({Name = "Aplicar tema Dragon Purple", Callback = function() redzlib:SetTheme("DarkPurple") end})
        studio:AddSection({"DESEMPENHO"})
        studio:AddToggle({Name = "Modo desempenho", Default = false, Callback = function(enabled)
            LyraParticlesEnabled = not enabled
            for _, particle in ipairs(LyraParticles) do if particle and particle.Parent then particle.Visible = not enabled end end
            if MainNeonGlow and MainNeonGlow.Parent then MainNeonGlow.Transparency = enabled and 0.82 or 0.56 end
        end})
        studio:AddButton({Name = "Restaurar visual padrão", Callback = function()
            redzlib:SetTheme("DarkPurple")
            r,g,b = 160,55,245; applyAccent()
            if BackgroundImage and BackgroundImage.Parent then BackgroundImage.ImageTransparency = 0.07 end
            if MainNeonGlow and MainNeonGlow.Parent then MainNeonGlow.Transparency = 0.56; MainNeonGlow.Enabled = true end
            LyraShineInterval = 3
            for _, particle in ipairs(LyraParticles) do if particle and particle.Parent then particle.Visible = true end end
        end})
    end
